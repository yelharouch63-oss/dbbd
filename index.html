function createFallingText() { const textNode = document.createElement('div'); textNode.classList.add('falling-text'); // اختيار نص ولون وسرعة وحجم عشوائي textNode.innerText = phrases[Math.floor(Math.random() * phrases.length)]; textNode.style.color = colors[Math.floor(Math.random() * colors.length)]; textNode.style.left = Math.random() * 95 + 'vw'; const fontSize = Math.random() * 18 + 14; // حجم الخط بين 14 و 32 textNode.style.fontSize = fontSize + 'px'; const duration = Math.random() * 4 + 4; // سرعة الهبوط (من 4 لـ 8 ثواني) textNode.style.animationDuration = duration + 's'; // إضافة تأثير الإضاءة (Glow) textNode.style.textShadow = `0 0 8px ${textNode.style.color}`; document.body.appendChild(textNode); // مسح العنصر بعد نهايته باش ما يتقالش المتصفح setTimeout(() => { textNode.remove(); }, duration * 1000); } // إنشاء كلمة جديدة كل 180 ميلي ثانية setInterval(createFallingText, 180);
